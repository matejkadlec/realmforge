# The pack's default keys (RLM-279), read by Default Options: key_<key mapping>:<key>[:<modifier>]. A player's
# own binding stays; a key still on its mod's default moves to the one here. Keys are named by their position on a
# US keyboard, while the game shows them as the player's layout labels them: on Matěj's Czech QWERTZ keyboard
# key.keyboard.y is labelled Z, and key.keyboard.grave.accent (left of 1) is labelled ;.

# Vanilla: Quick Actions and Swap Offhand are left free for spells (RLM-25). The spell bar (C3b, RLM-25) takes R, T, F,
# G, H and C, X for the racial ability and P for the spellbook (Matěj, 2026-10-15 and 2026-10-16), so chat moves from
# T to Enter, as in WoW, the creative toolbar keys (C and X) go unbound, and so does Social Interactions (P), which the
# pause menu's Player Reporting still opens.
key_key.quickActions:key.keyboard.unknown
key_key.swapOffhand:key.keyboard.unknown
key_key.chat:key.keyboard.enter
key_key.loadToolbarActivator:key.keyboard.unknown
key_key.saveToolbarActivator:key.keyboard.unknown
key_key.socialInteractions:key.keyboard.unknown

# Apotheosis: Open World Tier Select moves from Ctrl+T to Alt+T, so a player who holds Ctrl to sprint still casts the
# spell bar's T (RLM-25).
key_key.apotheosis.open_world_tier_select:key.keyboard.t:ALT

# FTB Quests: Open Quests on ; (Czech QWERTZ).
key_key.ftbquests.keys.quests:key.keyboard.grave.accent

# Inventory Sorting: Sort on ; too, beside Open Quests: it sorts while an inventory is open, Open Quests opens the book
# while none is (Matěj, 2026-10-19, RLM-308). Its default, P, is the spellbook's; so was Open Settings', now unbound
# (its settings stay in the mod list).
key_inventorysorter.key.sort:key.keyboard.grave.accent
key_inventorysorter.key.config:key.keyboard.unknown

# Tom's Simple Storage: Open Terminal on V (RLM-25).
key_key.toms_storage.open_terminal:key.keyboard.v

# Just Zoom: Zoom on Z (Czech QWERTZ; Xaero's Minimap Settings had that key); Open Options unbound (it shared B
# with Open Backpack).
key_justzoom.keybinds.keybind.zoom:key.keyboard.y
key_justzoom.keybinds.keybind.open_options:key.keyboard.unknown

# Vanilla: Open Command unbound (chat with a typed / does the same).
key_key.command:key.keyboard.unknown
